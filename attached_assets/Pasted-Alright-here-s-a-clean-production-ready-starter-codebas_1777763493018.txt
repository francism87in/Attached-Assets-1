# 🧱 2. Prisma Schema

## prisma/schema.prisma

```prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model User {
  id        String   @id @default(uuid())
  name      String
  email     String   @unique
  password  String
  role      String
  createdAt DateTime @default(now())
}

model Project {
  id              String   @id @default(uuid())
  name            String
  location        String
  city            String
  developerName   String
  possessionDate  DateTime?
  latitude        Float?
  longitude       Float?
  createdAt       DateTime @default(now())

  units           Unit[]
  scores          Score?
}

model Unit {
  id         String   @id @default(uuid())
  projectId  String
  type       String
  carpetArea Float
  price      Float
  floor      Int?
  available  Boolean @default(true)

  project    Project @relation(fields: [projectId], references: [id])
}

model Score {
  id               String  @id @default(uuid())
  projectId        String  @unique
  valueScore       Float
  locationScore    Float
  lifestyleScore   Float
  investmentScore  Float
  trustScore       Float
  overallScore     Float

  project          Project @relation(fields: [projectId], references: [id])
}
```

---

## project.controller.ts

```ts
import { Request, Response } from "express";
import * as service from "./project.service";

export const createProject = async (req: Request, res: Response) => {
  const data = await service.createProject(req.body);
  res.json(data);
};

export const getProjects = async (_req: Request, res: Response) => {
  const data = await service.getProjects();
  res.json(data);
};

export const getProjectById = async (req: Request, res: Response) => {
  const data = await service.getProjectById(req.params.id);
  res.json(data);
};
```

---

## project.service.ts

```ts
import { prisma } from "../../config/db";

export const createProject = async (data: any) => {
  return prisma.project.create({
    data,
  });
};

export const getProjects = async () => {
  return prisma.project.findMany({
    include: { units: true, scores: true },
  });
};

export const getProjectById = async (id: string) => {
  return prisma.project.findUnique({
    where: { id },
    include: { units: true, scores: true },
  });
};
```

# 🧮 7. Scoring Engine

## src/modules/scoring/scoring.service.ts

```ts
export const calculateScores = (project: any) => {
  const valueScore = 80; // placeholder logic
  const locationScore = 70;
  const lifestyleScore = project.units.length * 10;
  const investmentScore = 75;
  const trustScore = 60;

  const overallScore =
    (valueScore + locationScore + lifestyleScore + investmentScore + trustScore) /
    5;

  return {
    valueScore,
    locationScore,
    lifestyleScore,
    investmentScore,
    trustScore,
    overallScore,
  };
};
```

# 🤖 8. AI Module

## src/modules/ai/ai.service.ts

```ts
import axios from "axios";

export const generateSummary = async (project: any) => {
  const prompt = `
  Summarize this real estate project in 3 lines:
  ${JSON.stringify(project)}
  `;

  const response = await axios.post(
    "https://api.openai.com/v1/chat/completions",
    {
      model: "gpt-4o-mini",
      messages: [{ role: "user", content: prompt }],
    },
    {
      headers: {
        Authorization: `Bearer ${process.env.OPENAI_API_KEY}`,
      },
    }
  );

  return response.data.choices[0].message.content;
};
```

# 🌐 9. Public AI Endpoint

Add this inside project.routes.ts:

```ts
import { prisma } from "../../config/db";
import { generateSummary } from "../ai/ai.service";

router.get("/public/:id", async (req, res) => {
  const project = await prisma.project.findUnique({
    where: { id: req.params.id },
    include: { units: true, scores: true },
  });

  const summary = await generateSummary(project);

  res.json({
    ...project,
    ai_summary: summary,
  });
});
```
