# 🏢 7. PROJECT LIST PAGE

## app/projects/page.tsx

```tsx
"use client";

import { useEffect, useState } from "react";
import api from "@/services/api";
import Link from "next/link";

export default function Projects() {
  const [projects, setProjects] = useState<any[]>([]);

  useEffect(() => {
    api.get("/projects").then((res) => setProjects(res.data));
  }, []);

  return (
    <div>
      <h1 className="text-xl font-bold mb-4">Projects</h1>

      {projects.map((p) => (
        <Link key={p.id} href={`/projects/${p.id}`}>
          <div className="bg-white p-4 mb-3 rounded shadow cursor-pointer">
            {p.name}
          </div>
        </Link>
      ))}
    </div>
  );
}
```

# ➕ 8. ADD PROJECT PAGE

## app/projects/new/page.tsx

```tsx
"use client";

import { useState } from "react";
import api from "@/services/api";
import { useRouter } from "next/navigation";

export default function NewProject() {
  const router = useRouter();
  const [form, setForm] = useState({
    name: "",
    city: "",
    location: "",
  });

  const handleSubmit = async () => {
    await api.post("/projects", form);
    router.push("/projects");
  };

  return (
    <div className="max-w-lg">
      <h1 className="text-xl font-bold mb-4">Add Project</h1>

      <input
        className="border p-2 w-full mb-3"
        placeholder="Project Name"
        onChange={(e) => setForm({ ...form, name: e.target.value })}
      />

      <input
        className="border p-2 w-full mb-3"
        placeholder="City"
        onChange={(e) => setForm({ ...form, city: e.target.value })}
      />

      <input
        className="border p-2 w-full mb-3"
        placeholder="Location"
        onChange={(e) => setForm({ ...form, location: e.target.value })}
      />

      <button
        onClick={handleSubmit}
        className="bg-blue-600 text-white px-4 py-2 rounded"
      >
        Create Project
      </button>
    </div>
  );
}
```

# 📊 9. PROJECT DETAIL PAGE

## app/projects/[id]/page.tsx

```tsx
"use client";

import { useEffect, useState } from "react";
import api from "@/services/api";

export default function ProjectDetail({ params }: any) {
  const [project, setProject] = useState<any>(null);

  useEffect(() => {
    api.get(`/projects/${params.id}`).then((res) => setProject(res.data));
  }, []);

  if (!project) return <p>Loading...</p>;

  return (
    <div>
      <h1 className="text-2xl font-bold">{project.name}</h1>

      <div className="grid grid-cols-5 gap-4 mt-4">
        <div>Value: {project.scores?.valueScore}</div>
        <div>Location: {project.scores?.locationScore}</div>
        <div>Lifestyle: {project.scores?.lifestyleScore}</div>
        <div>Investment: {project.scores?.investmentScore}</div>
        <div>Trust: {project.scores?.trustScore}</div>
      </div>

      <div className="mt-6 bg-gray-100 p-4 rounded">
        API Endpoint:
        <pre>{`/public/project/${project.id}`}</pre>
      </div>
    </div>
  );
}
```

# 🌐 10. API CONSOLE

## app/api-console/page.tsx

```tsx
"use client";

import { useState } from "react";
import api from "@/services/api";

export default function ApiConsole() {
  const [response, setResponse] = useState<any>(null);
  const [id, setId] = useState("");

  const testApi = async () => {
    const res = await api.get(`/public/project/${id}`);
    setResponse(res.data);
  };

  return (
    <div>
      <h1 className="text-xl font-bold mb-4">API Console</h1>

      <input
        placeholder="Project ID"
        className="border p-2 mr-2"
        onChange={(e) => setId(e.target.value)}
      />

      <button onClick={testApi} className="bg-black text-white px-4 py-2">
        Test API
      </button>

      <pre className="mt-4 bg-gray-100 p-4">
        {JSON.stringify(response, null, 2)}
      </pre>
    </div>
  );
}
```
